using System;

class Program
{
    static void Main()
    {
        const string SumCommand = "sum";
        const string ExitCommand = "exit";
        const int InitialSize = 0;
        const int GrowthFactor = 1;

        int[] numbers = new int[InitialSize];
        int count = 0;
        bool isProgramRunning = true;

        Console.WriteLine("Введите числа или команды (sum, exit):");

        while (isProgramRunning)
        {
            if (count > 0)
            {
                Console.Write("Введенные числа: ");
                for (int i = 0; i < count; i++)
                {
                    Console.Write(numbers[i] + " ");
                }
                Console.WriteLine();
            }

            Console.Write("Ввод: ");
            string input = Console.ReadLine();

            if (input == ExitCommand)
            {
                Console.WriteLine("Выход из программы.");
                isProgramRunning = false;
            }
            else if (input == SumCommand)
            {
                int sum = 0;
                for (int i = 0; i < count; i++)
                {
                    sum += numbers[i];
                }
                Console.WriteLine($"Сумма всех чисел: {sum}");
            }
            else
            {
                if (int.TryParse(input, out int number))
                {
                    int newSize = numbers.Length + GrowthFactor;
                    int[] newArray = new int[newSize];

                    for (int i = 0; i < numbers.Length; i++)
                    {
                        newArray[i] = numbers[i];
                    }

                    numbers = newArray;
                    numbers[count] = number;
                    count++;
                    Console.WriteLine($"Число {number} добавлено");
                }
                else
                {
                    Console.WriteLine("Ошибка: введите число или команду sum/exit");
                }
            }

            Console.WriteLine();
        }
    }
}
