using System;

class Program
{
    static void Main()
    {
        const string CommandSum = "sum";
        const string CommandExit = "exit";
        const int ArraySizeIncrement = 1;

        int[] numbers = new int[0];
        bool isRunning = true;

        Console.WriteLine($"Введите числа или команды ({CommandSum}, {CommandExit}):");

        while (isRunning)
        {
            if (numbers.Length > 0)
            {
                Console.Write("Введенные числа: ");
                for (int i = 0; i < numbers.Length; i++)
                {
                    Console.Write(numbers[i] + " ");
                }
                Console.WriteLine();
            }

            Console.Write("Ввод: ");
            string input = Console.ReadLine();

            if (input == CommandExit)
            {
                Console.WriteLine("Выход из программы.");
                isRunning = false;
            }
            else if (input == CommandSum)
            {
                int sum = 0;
                for (int i = 0; i < numbers.Length; i++)
                {
                    sum += numbers[i];
                }
                Console.WriteLine($"Сумма всех чисел: {sum}");
            }
            else
            {
                if (int.TryParse(input, out int number))
                {
                    int newSize = numbers.Length + ArraySizeIncrement;
                    int[] newArray = new int[newSize];

                    for (int i = 0; i < numbers.Length; i++)
                    {
                        newArray[i] = numbers[i];
                    }

                    newArray[numbers.Length] = number;
                    numbers = newArray;
                    Console.WriteLine($"Число {number} добавлено");
                }
                else
                {
                    Console.WriteLine($"Ошибка: введите число или команду {CommandSum}/{CommandExit}");
                }
            }

            Console.WriteLine();
        }
    }
}
